@ ============================================================
@ DREAM GAME v0.9.2
@ ============================================================

scrn game

    @ --------------------------------------------------------
    @ PLAYER POSITION
    @ --------------------------------------------------------

    s x = 20
    s y = 20

    s dx = 1
    s dy = 0

    @ --------------------------------------------------------
    @ FOOD POSITION
    @ --------------------------------------------------------

    s foodx = rdm [1,55]
    s foody = rdm [1,40]

    @ Draw player
    pxl[x,y]

    @ Draw food
    pxl[foodx,foody]

    @ --------------------------------------------------------
    @ CONTROLS
    @ --------------------------------------------------------

    w [u-a] clk

        dy = -1
        dx = 0

    end

    w [d-a] clk

        dy = 1
        dx = 0

    end

    w [l-a] clk

        dx = -1
        dy = 0

    end

    w [r-a] clk

        dx = 1
        dy = 0

    end

    @ --------------------------------------------------------
    @ GAME LOOP
    @ --------------------------------------------------------

    rpt[frvr]

        @ Remove old player position
        clrr[x,y]

        @ Move player
        x = x + dx
        y = y + dy

        @ Draw player
        pxl[x,y]

        @ ----------------------------------------------------
        @ FOOD COLLISION
        @ ----------------------------------------------------

        if x == foodx

            if y == foody

                @ Remove eaten food
                clrr[foodx,foody]

                @ Generate new food position
                foodx = rdm [1,55]
                foody = rdm [1,40]

                @ Draw new food
                pxl[foodx,foody]

            end

        end

    end

end
